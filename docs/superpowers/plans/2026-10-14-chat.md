# Chat (Provider ↔ Traveler) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Realtime 1-1 chat between a traveler and a provider: text, reply, edit/delete (recall), listing link cards, images/files, shared-media history, location sharing, and starting a chat from a listing page.

**Architecture:** Extend the existing `Message` model (type, reply, edited/deleted, location, extracted links) plus a `MessageAttachment` table. New NestJS `messages` module: REST is the only write path; a Socket.IO gateway only pushes events to per-user rooms. FE replaces polling with a socket hook that patches the TanStack Query cache; chat UI is split into small Tailwind components under `fe/src/components/chat/`.

**Tech Stack:** NestJS 12, Prisma 6 (client `^6.19`), PostgreSQL, Socket.IO 4.8 (`@nestjs/websockets`, `@nestjs/platform-socket.io`), Supabase Storage (bucket `chat`), Next.js 16, TanStack Query v5, Zustand, React Hook Form + Zod 4, react-leaflet 5, Vitest + Testing Library, Playwright, Jest + supertest (BE).

Spec: `docs/superpowers/specs/2026-10-14-chat-design.md` (this plan refines it where noted in the "Spec deltas" list below).

### Spec deltas (decided while planning; Task 15 syncs them into the spec)

1. `Conversation.pairKey String? @unique` (sorted `userA:userB`) makes "start conversation" idempotent under races.
2. New `GET /messages/conversations/:id/info` returns the `Conversation` so the chat header works for a just-opened conversation that is hidden from the list (no messages yet).
3. `message:deleted` payload is `{ id, conversationId, deletedAt }`. The gateway also emits `ready` to a socket after it joined its user room (clients treat `ready` as "connected").
4. FE uses a `useChatSocket()` hook mounted by the messages page instead of an app-wide provider.
5. Location picker: click the map to place the pin (not drag).

- [ ] **Step 3: Implement `links.util.ts`**

```ts
const URL_RE = /https?:\/\/[^\s<>"']+/gi;
const TRAILING_PUNCTUATION = /[.,;:!?]+$/;
export const MAX_LINKS = 10;

export function extractLinks(content: string): string[] {
  const found = (content.match(URL_RE) ?? []).map((u) => u.replace(TRAILING_PUNCTUATION, ''));
  return [...new Set(found)].slice(0, MAX_LINKS);
}

export function listingIdFromUrl(raw: string, allowedOrigins: string[]): string | null {
  let url: URL;
  try {
    url = new URL(raw);
  } catch {
    return null;
  }
  if (!allowedOrigins.includes(url.origin)) return null;
  return /^\/listings\/([A-Za-z0-9_-]+)\/?$/.exec(url.pathname)?.[1] ?? null;
}

export function appOrigins(): string[] {
  const configured = (process.env.FRONTEND_URL ?? '').split(',');
  return ['http://localhost:3000', ...configured]
    .map((o) => o.trim())
    .filter(Boolean)
    .flatMap((o) => {
      try {
        return [new URL(o).origin];
      } catch {
        return [];
      }
    });
}
```

```ts
import { BadRequestException } from '@nestjs/common';

export const MAX_FILE_BYTES = 10 * 1024 * 1024;
export const MAX_FILES = 10;

type Kind = 'jpeg' | 'png' | 'gif' | 'webp' | 'pdf' | 'zip' | 'ole2' | 'text';

export const ALLOWED_MIME: Record<string, { kind: Kind; ext: string }> = {
  'image/jpeg': { kind: 'jpeg', ext: 'jpg' },
  'image/png': { kind: 'png', ext: 'png' },
  'image/gif': { kind: 'gif', ext: 'gif' },
  'image/webp': { kind: 'webp', ext: 'webp' },
  'application/pdf': { kind: 'pdf', ext: 'pdf' },
  'application/msword': { kind: 'ole2', ext: 'doc' },
  'application/vnd.openxmlformats-officedocument.wordprocessingml.document': { kind: 'zip', ext: 'docx' },
  'application/vnd.ms-excel': { kind: 'ole2', ext: 'xls' },
  'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet': { kind: 'zip', ext: 'xlsx' },
  'text/plain': { kind: 'text', ext: 'txt' },
  'application/zip': { kind: 'zip', ext: 'zip' },
};

export const isImageMime = (mime: string): boolean => mime.startsWith('image/');

const startsWith = (buf: Buffer, bytes: number[], offset = 0) =>
  bytes.every((b, i) => buf[offset + i] === b);

// The declared mime comes from the client, so the content signature is what we trust.
function sniff(buf: Buffer): Kind | null {
  if (startsWith(buf, [0xff, 0xd8, 0xff])) return 'jpeg';
  if (startsWith(buf, [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a])) return 'png';
  if (startsWith(buf, [0x47, 0x49, 0x46, 0x38]) && (buf[4] === 0x37 || buf[4] === 0x39) && buf[5] === 0x61) return 'gif';
  if (startsWith(buf, [0x52, 0x49, 0x46, 0x46]) && startsWith(buf, [0x57, 0x45, 0x42, 0x50], 8)) return 'webp';
  if (startsWith(buf, [0x25, 0x50, 0x44, 0x46, 0x2d])) return 'pdf';
  if (startsWith(buf, [0x50, 0x4b, 0x03, 0x04])) return 'zip';
  if (startsWith(buf, [0xd0, 0xcf, 0x11, 0xe0, 0xa1, 0xb1, 0x1a, 0xe1])) return 'ole2';
  return buf.subarray(0, 8000).includes(0) ? null : 'text';
}

export function validateAttachment(file: { buffer: Buffer; mimetype: string; size: number }): {
  mime: string;
  ext: string;
} {
  const entry = ALLOWED_MIME[file.mimetype];
  if (!entry) throw new BadRequestException('Định dạng file không được hỗ trợ');
  if (file.size === 0) throw new BadRequestException('File rỗng');
  if (file.size > MAX_FILE_BYTES) throw new BadRequestException('File vượt quá 10MB');
  if (sniff(file.buffer) !== entry.kind) throw new BadRequestException('Nội dung file không khớp định dạng');
  return { mime: file.mimetype, ext: entry.ext };
}

export function safeFileName(name: string): string {
  // eslint-disable-next-line no-control-regex
  const cleaned = name.replace(/[\\/]/g, '_').replace(/[\u0000-\u001f\u007f]/g, '').trim();
  return cleaned.slice(0, 200) || 'file';
}
```

```ts
import { Inject } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { JwtService } from '@nestjs/jwt';
import { OnGatewayConnection, WebSocketGateway, WebSocketServer } from '@nestjs/websockets';
import { Redis } from 'ioredis';
import { Server, Socket } from 'socket.io';
import { isAllowedOrigin } from '../../common/cors.util';
import { blockedKey } from '../../common/redis/auth-blocklist';
import { REDIS_CLIENT } from '../../common/redis/redis.module';

export type MessageEvent = 'message:new' | 'message:updated' | 'message:deleted' | 'message:read';

export const userRoom = (userId: string) => `user:${userId}`;

@WebSocketGateway({
  cors: {
    origin: (origin: string | undefined, cb: (err: Error | null, allow?: boolean) => void) =>
      cb(null, isAllowedOrigin(origin)),
    credentials: true,
  },
})
export class MessagesGateway implements OnGatewayConnection {
  @WebSocketServer() server!: Server;

  constructor(
    private readonly jwt: JwtService,
    private readonly config: ConfigService,
    @Inject(REDIS_CLIENT) private readonly redis: Redis,
  ) {}

  // The token is verified once at handshake; clients reconnect with a fresh token after expiry.
  async handleConnection(client: Socket): Promise<void> {
    try {
      const token: unknown = client.handshake.auth?.token;
      if (typeof token !== 'string') throw new Error('no token');
      const payload: { sub?: unknown } = await this.jwt.verifyAsync(token, {
        secret: this.config.getOrThrow<string>('JWT_ACCESS_SECRET'),
        algorithms: ['HS256'],
      });
      if (typeof payload.sub !== 'string' || (await this.redis.exists(blockedKey(payload.sub)))) {
        throw new Error('rejected');
      }
      await client.join(userRoom(payload.sub));
      client.emit('ready');
    } catch {
      client.disconnect(true);
    }
  }

  emitToUsers(userIds: string[], event: MessageEvent, payload: unknown): void {
    this.server.to(userIds.map(userRoom)).emit(event, payload);
  }
}
```

```ts
export function connect(url: string, token: string): Promise<Socket> {
  return new Promise((resolve, reject) => {
    const socket = io(url, { auth: { token }, transports: ['websocket'], reconnection: false });
    socket.once('ready', () => resolve(socket));
    socket.once('connect_error', reject);
  });

export function nextEvent<T = unknown>(socket: Socket, event: string, ms = 2000): Promise<T> {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => reject(new Error(`timed out waiting for ${event}`)), ms);
    socket.once(event, (payload: T) => {
      clearTimeout(timer);
      resolve(payload);
    });
  });
}
```

```ts
        { id: l.id, title: l.title, imageUrl: l.imageUrls[0] ?? null, price: Number(l.price), currency: l.currency, location: l.location },
      ]),
    );
  }

  /** messageId → previews, resolved with one DB query for the whole page. */
  async forMessages(
    rows: { id: string; links: string[]; deletedAt: Date | null }[],
  ): Promise<Map<string, ListingPreviewDto[]>> {
    const origins = appOrigins();
    const idsByMessage = new Map<string, string[]>();
    for (const r of rows) {
      if (r.deletedAt) continue;
      const ids = [...new Set(r.links.map((l) => listingIdFromUrl(l, origins)).filter((x): x is string => !!x))].slice(
        0,
        MAX_PREVIEWS_PER_MESSAGE,
      );
      if (ids.length) idsByMessage.set(r.id, ids);
    }
    const byId = await this.byListingIds([...new Set([...idsByMessage.values()].flat())]);
    const out = new Map<string, ListingPreviewDto[]>();
    for (const [messageId, ids] of idsByMessage) {
      const previews = ids.flatMap((id) => byId.get(id) ?? []);
      if (previews.length) out.set(messageId, previews);
    }
    return out;
  }

  /** url → preview, for the "Link" history tab. */
  async forUrls(urls: string[]): Promise<Map<string, ListingPreviewDto>> {
    const origins = appOrigins();
    const idByUrl = new Map<string, string>();
    for (const url of urls) {
      const id = listingIdFromUrl(url, origins);
      if (id) idByUrl.set(url, id);
    }
    const byId = await this.byListingIds([...new Set(idByUrl.values())]);
    const out = new Map<string, ListingPreviewDto>();
    for (const [url, id] of idByUrl) {
      const preview = byId.get(id);
      if (preview) out.set(url, preview);
    }
    return out;
  }
}
```

```ts
import { BadRequestException } from '@nestjs/common';
import { MessageType } from '@prisma/client';
import { ALLOWED_MIME, MAX_FILES, isImageMime } from './attachment-validator';
import type { SendMessageDto } from './messages.dto';

// class-validator cannot express "fields depend on `type`", so the invariants live here.
export function validateSendShape(dto: SendMessageDto): void {
  const attachments = dto.attachments ?? [];
  const hasLocation = dto.latitude !== undefined || dto.longitude !== undefined || dto.locationLabel !== undefined;

  switch (dto.type) {
    case MessageType.TEXT:
      if (!dto.content?.trim()) throw new BadRequestException('Nội dung tin nhắn không được để trống');
      if (attachments.length || hasLocation) throw new BadRequestException('Tin văn bản không kèm file hoặc vị trí');
      return;
    case MessageType.IMAGE:
    case MessageType.FILE: {
      if (attachments.length < 1 || attachments.length > MAX_FILES) {
        throw new BadRequestException(`Cần từ 1 đến ${MAX_FILES} file`);
      }
      if (hasLocation) throw new BadRequestException('Tin file không kèm vị trí');
      const wantImage = dto.type === MessageType.IMAGE;
      for (const a of attachments) {
        if (!ALLOWED_MIME[a.mimeType]) throw new BadRequestException('Định dạng file không được hỗ trợ');
        if (isImageMime(a.mimeType) !== wantImage) {
          throw new BadRequestException(wantImage ? 'Tin ảnh chỉ chứa ảnh' : 'Tin file không chứa ảnh');
        }
      }
      return;
    }
    case MessageType.LOCATION:
      if (dto.latitude === undefined || dto.longitude === undefined) throw new BadRequestException('Thiếu toạ độ');
      if (attachments.length) throw new BadRequestException('Tin vị trí không kèm file');
      return;
  }
}
```

```ts
import { BadRequestException, ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { MessageType, Prisma, Role, UserStatus } from '@prisma/client';
import { PrismaService } from '../../prisma/prisma.service';
import { LinkPreviewService } from './link-preview.service';
import { extractLinks } from './links.util';
import { MESSAGE_INCLUDE, MessageRow, toMessageDto } from './message.mapper';
import { validateSendShape } from './message-shape';
import { MessagesGateway } from './messages.gateway';
import type { SendMessageDto } from './messages.dto';
import type { ConversationDto, MessageDto, MessagePage } from './messages.types';

const MEMBER_USER = { select: { id: true, name: true, avatarUrl: true } } as const;

@Injectable()
export class MessagesService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly previews: LinkPreviewService,
    private readonly gateway: MessagesGateway,
  ) {}

  private async memberIds(conversationId: string): Promise<string[]> {
    const rows = await this.prisma.conversationMember.findMany({ where: { conversationId }, select: { userId: true } });
    return rows.map((r) => r.userId);
  }

  private async requireMember(userId: string, conversationId: string): Promise<string[]> {
    const ids = await this.memberIds(conversationId);
    if (!ids.includes(userId)) throw new ForbiddenException('Bạn không thuộc cuộc trò chuyện này');
    return ids;
  }

  private async hydrate(rows: MessageRow[]): Promise<MessageDto[]> {
    const previews = await this.previews.forMessages(rows);
    return rows.map((r) => toMessageDto(r, previews.get(r.id)));
  }

  private async loadConversations(userId: string, where: Prisma.ConversationWhereInput): Promise<ConversationDto[]> {
    const rows = await this.prisma.conversation.findMany({
      where: { members: { some: { userId } }, ...where },
      orderBy: { updatedAt: 'desc' },
      include: {
        members: { include: { user: MEMBER_USER } },
        messages: { orderBy: [{ createdAt: 'desc' }, { id: 'desc' }], take: 1, include: MESSAGE_INCLUDE },
      },
    });
    const unread = await this.prisma.message.groupBy({
      by: ['conversationId'],
      where: {
        conversationId: { in: rows.map((r) => r.id) },
        senderId: { not: userId },
        isRead: false,
        deletedAt: null,
      },
      _count: { _all: true },
    });
    const unreadBy = new Map(unread.map((u) => [u.conversationId, u._count._all]));
    const lastBy = new Map((await this.hydrate(rows.flatMap((r) => r.messages))).map((m) => [m.conversationId, m]));
    return rows.map((r) => ({
      id: r.id,
      createdAt: r.createdAt.toISOString(),
      updatedAt: r.updatedAt.toISOString(),
      members: r.members.map((m) => ({ userId: m.userId, joinedAt: m.joinedAt.toISOString(), user: m.user })),
      lastMessage: lastBy.get(r.id) ?? null,
      unreadCount: unreadBy.get(r.id) ?? 0,
    }));
  }

  async startConversation(userId: string, targetUserId: string): Promise<ConversationDto> {
    if (userId === targetUserId) throw new BadRequestException('Không thể chat với chính mình');
    const users = await this.prisma.user.findMany({
      where: { id: { in: [userId, targetUserId] } },
      select: { id: true, status: true, roles: { select: { role: true } } },
    });
    if (users.length !== 2) throw new NotFoundException('Người dùng không tồn tại');
    if (users.some((u) => u.status !== UserStatus.ACTIVE)) throw new BadRequestException('Tài khoản không khả dụng');
    const providers = users.filter((u) => u.roles.some((r) => r.role === Role.PROVIDER));
    if (providers.length !== 1) throw new BadRequestException('Chỉ có thể chat giữa traveler và provider');

    const pairKey = [userId, targetUserId].sort().join(':');
    const { id } = await this.prisma.conversation.upsert({
      where: { pairKey },
      update: {},
      create: { pairKey, members: { create: [{ userId }, { userId: targetUserId }] } },
      select: { id: true },
    });
    return this.conversationInfo(userId, id);
  }

  listConversations(userId: string): Promise<ConversationDto[]> {
    return this.loadConversations(userId, { messages: { some: {} } });
  }

  async conversationInfo(userId: string, conversationId: string): Promise<ConversationDto> {
    const [conversation] = await this.loadConversations(userId, { id: conversationId });
    if (!conversation) throw new ForbiddenException('Bạn không thuộc cuộc trò chuyện này');
    return conversation;
  }

  async listMessages(
    userId: string,
    conversationId: string,
    q: { cursor?: string; limit: number },
  ): Promise<MessagePage> {
    await this.requireMember(userId, conversationId);
    const rows = await this.prisma.message.findMany({
      where: { conversationId },
      orderBy: [{ createdAt: 'desc' }, { id: 'desc' }],
      take: q.limit + 1,
      ...(q.cursor ? { cursor: { id: q.cursor }, skip: 1 } : {}),
      include: MESSAGE_INCLUDE,
    });
    const hasMore = rows.length > q.limit;
    const page = hasMore ? rows.slice(0, q.limit) : rows;
    return { items: await this.hydrate(page), nextCursor: hasMore ? page[page.length - 1].id : null };
  }

  async send(userId: string, conversationId: string, dto: SendMessageDto): Promise<MessageDto> {
    const memberIds = await this.requireMember(userId, conversationId);
    validateSendShape(dto);
    if (dto.replyToId) {
      const target = await this.prisma.message.findUnique({ where: { id: dto.replyToId }, select: { conversationId: true } });
      if (target?.conversationId !== conversationId) {
        throw new BadRequestException('Tin nhắn được trả lời không thuộc cuộc trò chuyện này');
      }
    }
    const content = dto.content?.trim() ?? '';
    const row = await this.prisma.$transaction(async (tx) => {
      const created = await tx.message.create({
        data: {
          conversationId,
          senderId: userId,
          type: dto.type,
          content,
          replyToId: dto.replyToId,
          links: extractLinks(content),
          latitude: dto.type === MessageType.LOCATION ? dto.latitude : undefined,
          longitude: dto.type === MessageType.LOCATION ? dto.longitude : undefined,
          locationLabel: dto.type === MessageType.LOCATION ? dto.locationLabel : undefined,
        },
        include: MESSAGE_INCLUDE,
      });
      await tx.conversation.update({ where: { id: conversationId }, data: { updatedAt: new Date() } });
      return created;
    });
    const [message] = await this.hydrate([row]);
    this.gateway.emitToUsers(memberIds, 'message:new', message);
    return message;
  }

  async markRead(userId: string, conversationId: string): Promise<{ ok: true }> {
    const memberIds = await this.requireMember(userId, conversationId);
    const { count } = await this.prisma.message.updateMany({
      where: { conversationId, senderId: { not: userId }, isRead: false },
      data: { isRead: true },
    });
    if (count) this.gateway.emitToUsers(memberIds, 'message:read', { conversationId, readerId: userId });
    return { ok: true };
  }
}
```

```ts
  async uploadAttachments(
    userId: string,
    conversationId: string,
    files: Express.Multer.File[] | undefined,
  ): Promise<UploadedAttachmentDto[]> {
    await this.requireMember(userId, conversationId);
    if (!files?.length) throw new BadRequestException('Thiếu file đính kèm (field "files")');
    // Validate the whole batch first so a bad file never leaves orphans in storage.
    const validated = files.map((file) => ({ file, ...validateAttachment(file) }));
    return Promise.all(
      validated.map(async ({ file, mime, ext }) => ({
        url: await this.storage.upload(file.buffer, ext, mime),
        // multer decodes multipart filenames as latin1, which garbles UTF-8 names.
        name: safeFileName(Buffer.from(file.originalname, 'latin1').toString('utf8')),
        mimeType: mime,
        size: file.size,
      })),
    );
  }

  async shared(
    userId: string,
    conversationId: string,
    kind: 'media' | 'files' | 'links',
    q: { cursor?: string; limit: number },
  ): Promise<SharedPage<SharedAttachmentDto> | SharedPage<SharedLinkDto>> {
    await this.requireMember(userId, conversationId);
    return kind === 'links' ? this.sharedLinks(conversationId, q) : this.sharedAttachments(conversationId, kind, q);
  }

  private async sharedAttachments(
    conversationId: string,
    kind: 'media' | 'files',
    q: { cursor?: string; limit: number },
  ): Promise<SharedPage<SharedAttachmentDto>> {
    const rows = await this.prisma.messageAttachment.findMany({
      where: {
        message: { conversationId, deletedAt: null, type: kind === 'media' ? MessageType.IMAGE : MessageType.FILE },
      },
      orderBy: [{ message: { createdAt: 'desc' } }, { message: { id: 'desc' } }, { id: 'desc' }],
      take: q.limit + 1,
      ...(q.cursor ? { cursor: { id: q.cursor }, skip: 1 } : {}),
      include: { message: { select: { id: true, createdAt: true } } },
    });
    const hasMore = rows.length > q.limit;
    const page = hasMore ? rows.slice(0, q.limit) : rows;
    return {
      items: page.map((a) => ({
        id: a.id, url: a.url, name: a.name, mimeType: a.mimeType, size: a.size,
        messageId: a.message.id, createdAt: a.message.createdAt.toISOString(),
      })),
      nextCursor: hasMore ? page[page.length - 1].id : null,
    };
  }

  // Cursor is a message id: one page = up to `limit` messages, each expanded into its links.
  private async sharedLinks(
    conversationId: string,
    q: { cursor?: string; limit: number },
  ): Promise<SharedPage<SharedLinkDto>> {
    const rows = await this.prisma.message.findMany({
      where: { conversationId, deletedAt: null, links: { isEmpty: false } },
      orderBy: [{ createdAt: 'desc' }, { id: 'desc' }],
      take: q.limit + 1,
      ...(q.cursor ? { cursor: { id: q.cursor }, skip: 1 } : {}),
      select: { id: true, links: true, createdAt: true },
    });
    const hasMore = rows.length > q.limit;
    const page = hasMore ? rows.slice(0, q.limit) : rows;
    const previews = await this.previews.forUrls([...new Set(page.flatMap((m) => m.links))]);
    return {
      items: page.flatMap((m) =>
        m.links.map((url) => ({
          messageId: m.id, url, createdAt: m.createdAt.toISOString(), listingPreview: previews.get(url) ?? null,
        })),
      ),
      nextCursor: hasMore ? page[page.length - 1].id : null,
    };
  }
```

```ts
const URL_RE = /https?:\/\/[^\s<>"']+/gi;

export type TextPart = { kind: 'text' | 'link'; value: string };

export function splitLinks(text: string): TextPart[] {
  const parts: TextPart[] = [];
  let last = 0;
  for (const match of text.matchAll(URL_RE)) {
    const url = match[0].replace(/[.,;:!?]+$/, '');
    const start = match.index ?? 0;
    if (start > last) parts.push({ kind: 'text', value: text.slice(last, start) });
    parts.push({ kind: 'link', value: url });
    last = start + url.length;
  }
  if (last < text.length) parts.push({ kind: 'text', value: text.slice(last) });
  return parts;
}
```

```ts
import type { InfiniteData, QueryClient } from '@tanstack/react-query';
import type { Message, MessagePage } from '@/types/message';

type MessagesCache = InfiniteData<MessagePage, string | undefined>;

export const messagesKey = (conversationId: string) => ['messages', conversationId] as const;

export const SOCKET_EVENTS = ['message:new', 'message:updated', 'message:deleted', 'message:read'] as const;
export type SocketEvent = (typeof SOCKET_EVENTS)[number];

function mapItems(qc: QueryClient, conversationId: string, fn: (m: Message) => Message) {
  qc.setQueryData<MessagesCache>(messagesKey(conversationId), (old) =>
    old && { ...old, pages: old.pages.map((p) => ({ ...p, items: p.items.map(fn) })) },
  );
}

export function upsertMessage(qc: QueryClient, msg: Message) {
  qc.setQueryData<MessagesCache>(messagesKey(msg.conversationId), (old) => {
    if (!old?.pages.length) return old;
    let found = false;
    const pages = old.pages.map((p) => ({
      ...p,
      items: p.items.map((m) => {
        if (m.id !== msg.id) return m;
        found = true;
        return msg;
      }),
    }));
    if (!found) pages[0] = { ...pages[0], items: [msg, ...pages[0].items] };
    return { ...old, pages };
  });
}

export function markDeleted(qc: QueryClient, p: { id: string; conversationId: string; deletedAt: string }) {
  mapItems(qc, p.conversationId, (m) => {
    if (m.id === p.id) {
      return {
        ...m, content: null, attachments: [], latitude: null, longitude: null, locationLabel: null,
        listingPreviews: [], deletedAt: p.deletedAt,
      };
    }
    if (m.replyTo?.id === p.id) return { ...m, replyTo: { ...m.replyTo, content: null, deleted: true } };
    return m;
  });
}

export function markRead(qc: QueryClient, p: { conversationId: string; readerId: string }) {
  mapItems(qc, p.conversationId, (m) => (m.senderId !== p.readerId && !m.isRead ? { ...m, isRead: true } : m));
}

export function applySocketEvent(qc: QueryClient, event: SocketEvent, payload: unknown) {
  const { conversationId } = payload as { conversationId: string };
  if (event === 'message:new' || event === 'message:updated') upsertMessage(qc, payload as Message);
  else if (event === 'message:deleted') markDeleted(qc, payload as { id: string; conversationId: string; deletedAt: string });
  else markRead(qc, payload as { conversationId: string; readerId: string });
  qc.invalidateQueries({ queryKey: ['conversations'] });
  qc.invalidateQueries({ queryKey: ['shared', conversationId] });
}
```

```ts
      // refresh the token through a normal API call, then reconnect.
      // ponytail: retries every 3s while mounted; add backoff if auth outages become common.
      if (reason === 'io server disconnect') {
        retry = setTimeout(() => void userApi.me().catch(() => undefined).finally(() => socket.connect()), RETRY_MS);
      }
    });
    for (const event of SOCKET_EVENTS) socket.on(event, (payload) => applySocketEvent(qc, event, payload));
    socket.connect();

    return () => {
      clearTimeout(retry);
      socket.removeAllListeners();
      socket.disconnect();
      setConnected(false);
    };
  }, [isAuthenticated, qc, setConnected]);
}
```

```tsx
'use client';

import { CornerUpLeft, FileText, Pencil, Trash2 } from 'lucide-react';
import { useState } from 'react';
import { cn } from 'cn';
import { formatBytes, formatTime, previewText } from '@/lib/chat/format';
import { splitLinks } from '@/lib/chat/links';
import type { Message } from '@/types/message';
import { ListingPreviewCard } from './listing-preview-card';
import { LocationCard } from './location-card';

interface Props {
  message: Message;
  isMine: boolean;
  onReply: (message: Message) => void;
  onEdit: (id: string, content: string) => Promise<unknown>;
  onDelete: (id: string) => Promise<unknown>;
  onJumpTo: (id: string) => void;
}

const actionClass =
  'rounded p-1 text-[#8694ad] opacity-0 transition-opacity hover:bg-[#e2e8f3] focus-visible:opacity-100 group-hover:opacity-100';

export function MessageBubble({ message: m, isMine, onReply, onEdit, onDelete, onJumpTo }: Props) {
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(m.content ?? '');
  const [error, setError] = useState<string | null>(null);
  const deleted = m.deletedAt !== null;

  async function run(action: () => Promise<unknown>, onDone?: () => void) {
    try {
      await action();
      setError(null);
      onDone?.();
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Thao tác thất bại');
    }
  }

  const save = () => run(() => onEdit(m.id, draft), () => setEditing(false));
  const remove = () => {
    if (window.confirm('Thu hồi tin nhắn này?')) void run(() => onDelete(m.id));
  };

  return (
    <div
      id={`msg-${m.id}`}
      data-testid="message"
      className={cn('group flex items-end gap-1', isMine ? 'flex-row-reverse' : 'flex-row')}
    >
      <div className={cn('flex max-w-[70%] flex-col', isMine ? 'items-end' : 'items-start')}>
        <div
          className={cn(
            'rounded-2xl px-3.5 py-2 text-sm',
            isMine ? 'bg-[#2a7bd9] text-white' : 'border border-[#DCEFFB] bg-white text-[#0f1729]',
            deleted && 'bg-transparent italic text-[#8694ad] ring-1 ring-[#DCEFFB]',
          )}
        >
          {deleted ? (
            'Tin nhắn đã bị thu hồi'
          ) : (
            <>
              {m.replyTo && (
                <button
                  type="button"
                  onClick={() => onJumpTo(m.replyTo!.id)}
                  className="mb-1 block w-full truncate rounded border-l-2 border-current bg-black/10 px-2 py-1 text-left text-xs opacity-80"
                >
                  {previewText({ type: m.replyTo.type, content: m.replyTo.content, deleted: m.replyTo.deleted })}
                </button>
              )}
              {editing ? (
                <div className="flex flex-col gap-1.5 text-[#0f1729]">
                  <textarea
                    aria-label="Sửa tin nhắn"
                    value={draft}
                    onChange={(e) => setDraft(e.target.value)}
                    className="w-64 rounded border border-[#DCEFFB] p-1.5 text-sm"
                  />
                  <div className="flex justify-end gap-2 text-xs">
                    <button type="button" onClick={() => { setEditing(false); setDraft(m.content ?? ''); }}>Huỷ</button>
                    <button type="button" onClick={save} className="font-semibold text-[#2a7bd9]">Lưu</button>
                  </div>
                </div>
              ) : (
                <MessageBody message={m} />
              )}
            </>
          )}
        </div>
        {!deleted && m.listingPreviews.map((p) => <ListingPreviewCard key={p.id} preview={p} />)}
        {error && <p role="alert" className="mt-1 text-xs text-red-500">{error}</p>}
        <p className="mt-0.5 text-[10px] text-[#8694ad]">
          {formatTime(m.createdAt)}
          {m.editedAt && !deleted && <> · <span>đã chỉnh sửa</span></>}
        </p>
      </div>
      {!deleted && (
        <div className="flex shrink-0 pb-4">
          <button type="button" aria-label="Trả lời" onClick={() => onReply(m)} className={actionClass}>
            <CornerUpLeft className="h-4 w-4" />
          </button>
          {isMine && m.type === 'TEXT' && (
            <button type="button" aria-label="Sửa" onClick={() => setEditing(true)} className={actionClass}>
              <Pencil className="h-4 w-4" />
            </button>
          )}
          {isMine && (
            <button type="button" aria-label="Xoá" onClick={remove} className={actionClass}>
              <Trash2 className="h-4 w-4" />
            </button>
          )}
        </div>
      )}
    </div>
  );
}

function MessageBody({ message: m }: { message: Message }) {
  return (
    <>
      {m.type === 'IMAGE' && (
        <div className="mb-1 grid grid-cols-2 gap-1">
          {m.attachments.map((a) => (
            <a key={a.id} href={a.url} target="_blank" rel="noopener noreferrer">
              <img src={a.url} alt={a.name} className="max-h-48 rounded-lg object-cover" />
            </a>
          ))}
        </div>
      )}
      {m.type === 'FILE' &&
        m.attachments.map((a) => (
          <a
            key={a.id}
            href={a.url}
            target="_blank"
            rel="noopener noreferrer"
            download={a.name}
            className="mb-1 flex items-center gap-2 rounded-lg bg-black/10 px-2 py-1.5"
          >
            <FileText className="h-5 w-5 shrink-0" />
            <span className="min-w-0">
              <span className="block truncate text-sm font-medium">{a.name}</span>
              <span className="block text-xs opacity-70">{formatBytes(a.size)}</span>
            </span>
          </a>
        ))}
      {m.type === 'LOCATION' && m.latitude !== null && m.longitude !== null && (
        <LocationCard latitude={m.latitude} longitude={m.longitude} label={m.locationLabel} />
      )}
      {m.content && (
        <p className="whitespace-pre-wrap break-words">
          {splitLinks(m.content).map((part, i) =>
            part.kind === 'link' ? (
              <a key={i} href={part.value} target="_blank" rel="noopener noreferrer" className="underline">
                {part.value}
              </a>
            ) : (
              <span key={i}>{part.value}</span>
            ),
          )}
        </p>
      )}
    </>
  );
}
```

```tsx
interface Props {

  const [position, setPosition] = useState<Point | null>(null);

  const [error, setError] = useState<string | null>(null);
```

```tsx
'use client';

import { zodResolver } from '@hookform/resolvers/zod';
import { MapPin, Paperclip, Send, X } from 'lucide-react';
import { useEffect, useRef, useState } from 'react';
import { useForm } from 'react-hook-form';
import { previewText } from '@/lib/chat/format';
import { ACCEPT, textMessageSchema, validateFiles } from '@/lib/validators/message';
import type { AttachmentInput, Message, SendMessageInput } from '@/types/message';
import { LocationPicker } from './location-picker';

interface Props {
  replyTo: Message | null;
  onCancelReply: () => void;
  initialText?: string;
  onSend: (input: SendMessageInput) => Promise<void>;
  onUpload: (files: File[]) => Promise<AttachmentInput[]>;
}

const iconButton = 'rounded-lg p-2 text-[#5a6b85] hover:bg-[#e2e8f3] disabled:opacity-50';

export function MessageComposer({ replyTo, onCancelReply, initialText, onSend, onUpload }: Props) {
  const { register, getValues, setValue, watch, reset } = useForm<{ content: string }>({
    defaultValues: { content: initialText ?? '' },
  });
  const [files, setFiles] = useState<File[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [sending, setSending] = useState(false);
  const [pickerOpen, setPickerOpen] = useState(false);
  const fileInput = useRef<HTMLInputElement>(null);
  const content = watch('content');

  useEffect(() => {
    if (initialText && !getValues('content')) setValue('content', initialText);
  }, [initialText, getValues, setValue]);

  function pickFiles(selected: FileList | null) {
    const next = [...files, ...Array.from(selected ?? [])];
    const problem = validateFiles(next);
    setError(problem);
    if (!problem) setFiles(next);
    if (fileInput.current) fileInput.current.value = '';
  }

  async function submit() {
    const text = textMessageSchema.safeParse({ content: getValues('content') });
    const caption = text.success ? text.data.content : '';
    if (!files.length && !text.success) return;
    setSending(true);
    setError(null);
    try {
      const replyToId = replyTo?.id;
      if (!files.length) {
        await onSend({ type: 'TEXT', content: caption, replyToId });
      } else {
        const groups = [
          { type: 'IMAGE' as const, items: files.filter((f) => f.type.startsWith('image/')) },
          { type: 'FILE' as const, items: files.filter((f) => !f.type.startsWith('image/')) },
        ].filter((g) => g.items.length);
        for (const [i, group] of groups.entries()) {
          const attachments = await onUpload(group.items);
          await onSend({ type: group.type, content: i === 0 ? caption : '', replyToId: i === 0 ? replyToId : undefined, attachments });
        }
      }
      reset({ content: '' });
      setFiles([]);
      onCancelReply();
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Gửi thất bại, thử lại');
    } finally {
      setSending(false);
    }
  }

  async function sendLocation(value: { latitude: number; longitude: number; locationLabel?: string }) {
    setError(null);
    try {
      await onSend({ type: 'LOCATION', replyToId: replyTo?.id, ...value });
      onCancelReply();
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Gửi vị trí thất bại');
    }
  }

  const canSend = !sending && (files.length > 0 || content.trim().length > 0);

  return (
    <div className="border-t border-[#DCEFFB] bg-white p-3">
      {replyTo && (
        <div className="mb-2 flex items-center justify-between rounded-lg bg-[#f6fbfe] px-3 py-1.5 text-xs">
          <span className="truncate">
            Đang trả lời <b>{replyTo.sender.name}</b>: {previewText({ type: replyTo.type, content: replyTo.content })}
          </span>
          <button type="button" aria-label="Huỷ trả lời" onClick={onCancelReply}><X className="h-4 w-4" /></button>
        </div>
      )}
      {files.length > 0 && (
        <ul className="mb-2 flex flex-wrap gap-2">
          {files.map((f, i) => (
            <li key={`${f.name}-${i}`} className="flex items-center gap-1 rounded-full bg-[#e2e8f3] px-2.5 py-1 text-xs">
              <span className="max-w-32 truncate">{f.name}</span>
              <button type="button" aria-label={`Bỏ ${f.name}`} onClick={() => setFiles(files.filter((_, j) => j !== i))}>
                <X className="h-3 w-3" />
              </button>
            </li>
          ))}
        </ul>
      )}
      {error && <p role="alert" className="mb-2 text-xs text-red-500">{error}</p>}
      <div className="flex items-end gap-1">
        <input ref={fileInput} type="file" multiple accept={ACCEPT} aria-label="Chọn file" className="hidden" onChange={(e) => pickFiles(e.target.files)} />
        <button type="button" aria-label="Đính kèm" className={iconButton} onClick={() => fileInput.current?.click()}><Paperclip className="h-5 w-5" /></button>
        <button type="button" aria-label="Chia sẻ vị trí" className={iconButton} onClick={() => setPickerOpen(true)}><MapPin className="h-5 w-5" /></button>
        <textarea
          aria-label="Nhập tin nhắn"
          rows={1}
          placeholder="Nhập tin nhắn..."
          className="max-h-32 min-h-10 flex-1 resize-none rounded-xl border border-[#DCEFFB] px-3 py-2 text-sm outline-none focus:border-[#2a7bd9]"
          {...register('content')}
          onKeyDown={(e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
              e.preventDefault();
              void submit();
            }
          }}
        />
        <button type="button" aria-label="Gửi" disabled={!canSend} onClick={() => void submit()} className="rounded-xl bg-[#2a7bd9] p-2.5 text-white disabled:opacity-50">
          <Send className="h-5 w-5" />
        </button>
      </div>
      <LocationPicker open={pickerOpen} onOpenChange={setPickerOpen} onSubmit={sendLocation} />
    </div>
  );
}
```

```tsx
'use client';

import { useEffect, useRef } from 'react';
import type { Message } from '@/types/message';
import { MessageBubble } from './message-bubble';

interface Props {
  messages: Message[]; // newest first, as returned by the API
  currentUserId: string;
  hasMore: boolean;
  isFetchingMore: boolean;
  onLoadMore: () => void;
  jumpToId: string | null;
  onJumpHandled: () => void;
  onReply: (m: Message) => void;
  onEdit: (id: string, content: string) => Promise<unknown>;
  onDelete: (id: string) => Promise<unknown>;
  onJumpTo: (id: string) => void;
}

export function MessageList({
  messages, currentUserId, hasMore, isFetchingMore, onLoadMore, jumpToId, onJumpHandled, onReply, onEdit, onDelete, onJumpTo,
}: Props) {
  const bottomRef = useRef<HTMLDivElement>(null);
  const newestId = messages[0]?.id;

  // Follow new messages, but not while loading older pages (newestId stays the same then).
  useEffect(() => {
    bottomRef.current?.scrollIntoView({ block: 'end' });
  }, [newestId]);

  // A jump target may be on an older page: keep loading pages until it is rendered.
  useEffect(() => {
    if (!jumpToId) return;
    const el = document.getElementById(`msg-${jumpToId}`);
    if (el) {
      el.scrollIntoView({ block: 'center', behavior: 'smooth' });
      el.classList.add('ring-2', 'ring-[#92eee8]', 'rounded-2xl');
      setTimeout(() => el.classList.remove('ring-2', 'ring-[#92eee8]', 'rounded-2xl'), 1500);
      onJumpHandled();
    } else if (hasMore && !isFetchingMore) {
      onLoadMore();
    } else if (!hasMore && !isFetchingMore) {
      onJumpHandled();
    }
  }, [jumpToId, messages, hasMore, isFetchingMore, onLoadMore, onJumpHandled]);

  return (
    <div role="log" aria-label="Tin nhắn" className="flex flex-1 flex-col gap-3 overflow-y-auto bg-[#f6fbfe] p-5">
      {hasMore && (
        <button type="button" onClick={onLoadMore} disabled={isFetchingMore} className="self-center text-xs text-[#2a7bd9]">
          {isFetchingMore ? 'Đang tải...' : 'Xem tin nhắn cũ hơn'}
        </button>
      )}
      {[...messages].reverse().map((m) => (
        <MessageBubble
          key={m.id}
          message={m}
          isMine={m.senderId === currentUserId}
          onReply={onReply}
          onEdit={onEdit}
          onDelete={onDelete}
          onJumpTo={onJumpTo}
        />
      ))}
      <div ref={bottomRef} />
    </div>
  );
}
```

```tsx
interface Props {
  conversation: Conversation | undefined;
  conversationId: string;
  currentUserId: string;
  initialText?: string;
}

export function ChatWindow({ conversation, conversationId, currentUserId, initialText }: Props) {
  const { data, hasNextPage, fetchNextPage, isFetchingNextPage } = useMessages(conversationId);
  const send = useSendMessage(conversationId);
  const edit = useEditMessage();
  const remove = useDeleteMessage(conversationId);
  const upload = useUploadAttachments(conversationId);
  const markRead = useMarkRead(conversationId);
  const [replyTo, setReplyTo] = useState<Message | null>(null);
  const [showShared, setShowShared] = useState(false);
  const [jumpToId, setJumpToId] = useState<string | null>(null);

  const messages = data?.pages.flatMap((p) => p.items) ?? [];
  const other = conversation?.members.find((m) => m.userId !== currentUserId)?.user;

  // Mark read when someone else's unread message is on screen; once per newest unread id.
  const lastMarked = useRef<string | null>(null);
  const newestUnread = messages.find((m) => m.senderId !== currentUserId && !m.isRead)?.id ?? null;
  useEffect(() => {
    if (newestUnread && newestUnread !== lastMarked.current) {
      lastMarked.current = newestUnread;
      markRead.mutate();
    }
  }, [newestUnread, markRead]);

  const handleLoadMore = useCallback(() => void fetchNextPage(), [fetchNextPage]);
  const handleJumpHandled = useCallback(() => setJumpToId(null), []);

  async function handleSend(input: SendMessageInput) {
    await send.mutateAsync(input);
  }

  return (
    <div className="flex min-w-0 flex-1">
      <div className="flex min-w-0 flex-1 flex-col">
        <div className="flex items-center gap-3 border-b border-[#DCEFFB] bg-white px-5 py-3.5">

            <Images className="h-5 w-5" />
          </button>
        </div>
        <MessageList
          messages={messages}
          currentUserId={currentUserId}
          hasMore={!!hasNextPage}
          isFetchingMore={isFetchingNextPage}
          onLoadMore={handleLoadMore}
          jumpToId={jumpToId}
          onJumpHandled={handleJumpHandled}
          onReply={setReplyTo}
          onEdit={(id, content) => edit.mutateAsync({ id, content })}
          onDelete={(id) => remove.mutateAsync(id)}
          onJumpTo={setJumpToId}

          initialText={initialText}

          onUpload={(files) => upload.mutateAsync(files)}

      {showShared && (
```

```tsx
'use client';

import { usePathname, useRouter } from 'next/navigation';
import { useStartConversation } from '@/hooks/useMessages';
import { useAuthStore } from '@/store/auth.store';

export function ContactProviderButton({ providerId, listingId }: { providerId: string; listingId: string }) {
  const router = useRouter();
  const pathname = usePathname();
  const { user, isAuthenticated } = useAuthStore();
  const start = useStartConversation();
  const roles = user?.roles ?? [];

  // The BE only allows traveler↔provider chats, so don't offer the button to anyone else.
  if (isAuthenticated && (!roles.includes('TRAVELER') || roles.includes('PROVIDER') || user?.id === providerId)) {
    return null;
  }

  async function open() {
    if (!isAuthenticated) {
      router.push(`/login?redirect=${encodeURIComponent(pathname)}`);
      return;
    }
    const conversation = await start.mutateAsync(providerId).catch(() => null);
    if (conversation) router.push(`/messages?c=${conversation.id}&listing=${listingId}`);
  }

  return (
    <>
      <button
        type="button"
        onClick={open}
        disabled={start.isPending}
        className="mt-3 inline-flex items-center gap-2 rounded-lg border border-[#DCEFFB] px-4 py-2 text-sm font-medium text-[#2a7bd9] transition-all hover:shadow-sm disabled:opacity-60"
      >
        💬 Liên hệ Provider
      </button>
      {start.error && <p role="alert" className="mt-1 text-xs text-red-500">{start.error.message}</p>}
    </>
  );
}
```
